cmake_minimum_required(VERSION 3.22)
project(Flysnapper VERSION 0.4.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(FLYSNAPPER_CORE_ONLY "Build dependency-free MIDI engine tests" OFF)
option(FLYSNAPPER_BUILD_TESTS "Build validation tools" ON)

add_library(FlysnapperCore STATIC Source/Core/MusicData.cpp Source/Core/MidiEngine.cpp)
target_include_directories(FlysnapperCore PUBLIC Source)
if(MSVC)
  target_compile_options(FlysnapperCore PRIVATE /W4 /utf-8)
else()
  target_compile_options(FlysnapperCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FLYSNAPPER_BUILD_TESTS)
  enable_testing()
  add_executable(FlysnapperCoreTests Tests/CoreTests.cpp)
  target_link_libraries(FlysnapperCoreTests PRIVATE FlysnapperCore)
  add_test(NAME FlysnapperCore COMMAND FlysnapperCoreTests)
  set_tests_properties(FlysnapperCore PROPERTIES TIMEOUT 120)
  if(MSVC)
    target_compile_options(FlysnapperCoreTests PRIVATE /utf-8)
    target_link_options(FlysnapperCoreTests PRIVATE /STACK:8388608)
  endif()
endif()
if(FLYSNAPPER_CORE_ONLY)
  return()
endif()
set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps/JUCE" CACHE PATH "Verified JUCE 8.0.16-security checkout")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Run python build_plugin.py to acquire pinned JUCE and build.")
endif()
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/juce")
juce_add_plugin(Flysnapper
  VERSION 0.4.3 COMPANY_NAME "Flysnapper" BUNDLE_ID "audio.flysnapper.midi"
  PLUGIN_MANUFACTURER_CODE Flys PLUGIN_CODE Fsm1 PRODUCT_NAME "Flysnapper MIDI"
  DESCRIPTION "MIDI chord-tone and chord-scale correction with silent instrument output"
  IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE IS_MIDI_EFFECT FALSE
  EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE
  VST3_CATEGORIES Instrument FORMATS VST3)
target_sources(Flysnapper PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_include_directories(Flysnapper PUBLIC Source)
target_compile_definitions(Flysnapper PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Flysnapper PRIVATE FlysnapperCore juce::juce_audio_processors juce::juce_gui_extra
  PUBLIC juce::juce_recommended_config_flags)
if(MSVC)
  target_compile_options(Flysnapper PRIVATE /utf-8)
endif()
if(FLYSNAPPER_BUILD_TESTS)
  foreach(test IN ITEMS FlysnapperPluginTests FlysnapperMidiOnlyProcessorTests)
    juce_add_console_app(${test} PRODUCT_NAME "${test}")
    if(test STREQUAL "FlysnapperPluginTests")
      target_sources(${test} PRIVATE Tests/PluginTests.cpp)
    else()
      target_sources(${test} PRIVATE Tests/MidiOnlyProcessorTests.cpp)
    endif()
    target_link_libraries(${test} PRIVATE Flysnapper FlysnapperCore juce::juce_audio_processors juce::juce_gui_extra)
    target_compile_definitions(${test} PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    if(MSVC)
      target_compile_options(${test} PRIVATE /utf-8)
      target_link_options(${test} PRIVATE /STACK:8388608)
    endif()
  endforeach()
  add_test(NAME FlysnapperPluginSmoke COMMAND FlysnapperPluginTests)
  add_test(NAME FlysnapperMidiOnlyProcessor COMMAND FlysnapperMidiOnlyProcessorTests)
  set_tests_properties(FlysnapperPluginSmoke FlysnapperMidiOnlyProcessor PROPERTIES TIMEOUT 180)

  juce_add_console_app(FlysnapperVst3HostTests PRODUCT_NAME "Flysnapper actual MIDI-only VST3 tests")
  target_sources(FlysnapperVst3HostTests PRIVATE Tests/Vst3HostTests.cpp)
  target_include_directories(FlysnapperVst3HostTests PRIVATE "${JUCE_DIR}/modules/juce_audio_processors_headless/format_types/VST3_SDK")
  target_link_libraries(FlysnapperVst3HostTests PRIVATE juce::juce_audio_processors juce::juce_gui_extra juce::juce_recommended_config_flags)
  target_compile_definitions(FlysnapperVst3HostTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_VST3_CAN_REPLACE_VST2=0)
  if(MSVC)
    target_compile_options(FlysnapperVst3HostTests PRIVATE /utf-8)
    target_link_options(FlysnapperVst3HostTests PRIVATE /STACK:8388608)
  endif()
  add_dependencies(FlysnapperVst3HostTests Flysnapper_VST3)
  add_test(NAME FlysnapperActualVst3 COMMAND FlysnapperVst3HostTests "${CMAKE_BINARY_DIR}/Flysnapper_artefacts/$<CONFIG>/VST3/Flysnapper MIDI.vst3")
  set_tests_properties(FlysnapperActualVst3 PROPERTIES TIMEOUT 180)
endif()
